// ดึงข้อมูลย้อนหลัง (เช่น 100-200 จุดล่าสุด) จาก ThingSpeak const channelID = "ใส่เลข_CHANNEL_ID_ของคุณ"; const readKey = "ใส่_READ_API_KEY_ถ้ามี"; // ถ้าเป็น Public Channel เว้นว่างหรือใส่แค่ URL ได้ const url = `https://api.thingspeak.com/channels/${channelID}/feeds.json?results=200`; fetch(url) .then(response => response.json()) .then(data => { const feeds = data.feeds; if (!feeds || feeds.length === 0) return; // 1. นำค่าล่าสุดไปแสดงในการ์ดระดับน้ำปัจจุบัน const lastFeed = feeds[feeds.length - 1]; const currentVal = parseFloat(lastFeed.field1).toFixed(2); document.getElementById("current-water-level").innerText = `${currentVal} ซม.`; // (อัปเดตความสูงแท่งน้ำจำลองตาม currentVal ด้วย) // 2. จัดกลุ่มข้อมูลแยกตามชั่วโมง (เหมือนหน้า history.html) const hourlyGroups = {}; feeds.forEach(item => { if (!item.field1) return; const date = new Date(item.created_at); // สร้าง Key ชั่วโมง เช่น "10:00 น.", "11:00 น." const hourKey = date.getHours().toString().padStart(2, '0') + ":00 น."; if (!hourlyGroups[hourKey]) { hourlyGroups[hourKey] = []; } hourlyGroups[hourKey].push(parseFloat(item.field1)); }); // 3. คำนวณค่าเฉลี่ยของแต่ละชั่วโมง const hours = []; const averages = []; const tableData = []; for (const [hour, values] of Object.entries(hourlyGroups)) { const sum = values.reduce((a, b) => a + b, 0); const avg = (sum / values.length).toFixed(2); hours.push(hour); averages.push(avg); tableData.push({ hour, avg }); } // 4. นำ tableData ไปเรนเดอร์ลงในตารางเฉลี่ยรายชั่วโมง (ซ้ายล่าง) renderHourlyTable(tableData); // 5. นำ hours และ averages ไปอัปเดตกราฟ Chart.js (ตรงกลาง) updateChart(hours, averages); }) .catch(err => console.error("Error loading data:", err));
